export const smithyCode = `\$version: "2"
namespace example.hello

use alloy#simpleRestJson

@simpleRestJson
service HelloService {
    operations: [SayHello]
}

@http(method: "GET", uri: "/hello/{name}")
@readonly
operation SayHello {
    input := {
        @required @httpLabel
        name: String
    }
    output := {
        @required
        message: String
    }
}`;

export const serverCode = `// Generated by NSmithy:
public interface IHelloServiceHandler
{
    Task<SayHelloOutput> SayHelloAsync(
        SayHelloInput input,
        CancellationToken ct = default);
}

// You implement it:
public class HelloHandler : IHelloServiceHandler
{
    public Task<SayHelloOutput> SayHelloAsync(
        SayHelloInput input, CancellationToken ct = default) =>
        Task.FromResult(new SayHelloOutput("Hello, " + input.Name + "!"));
}

// Wire up in Program.cs:
builder.Services.AddSmithyServer<IHelloServiceHandler, HelloHandler>();
app.MapHelloServiceRoutes();`;

export const clientCode = `var client = new HelloServiceClient(
    transport,
    new SmithyClientOptions { Endpoint = new Uri("https://api.example.com") }
);

var response = await client.SayHelloAsync(new SayHelloInput("World"));
Console.WriteLine(response.Message); // Hello, World!`;

### Define your contract

Start by describing your API in [Smithy IDL](https://smithy.io) — operations, input and output shapes, and traits. Smithy is protocol-agnostic: the same model can drive a REST service, a gRPC service, or both at once.

</div>

</div>

<div class="how-it-works">
<div>

### Implement server stubs

NSmithy generates a typed handler interface — one method per service operation. You fill in the business logic; routing, serialization, and protocol binding are handled by the generated ASP.NET Core adapter.

</div>
<Code code={serverCode} lang="csharp" />
</div>

<div class="how-it-works">
<div>

### Use the generated client

NSmithy generates a typed async client with one method per operation. Inputs and outputs are strongly-typed C# records — no string URLs, no manual JSON, no reflection.

</div>
<Code code={clientCode} lang="csharp" />
</div>
