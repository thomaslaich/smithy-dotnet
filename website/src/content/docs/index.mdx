export const smithyCode = `\$version: "2"
namespace example.hello

use alloy#simpleRestJson

@simpleRestJson
service HelloService {
    operations: [SayHello]
}

@http(method: "GET", uri: "/hello/{name}")
@readonly
operation SayHello {
    input := {
        @required
        @httpLabel
        name: String
    }
    output := {
        @required
        message: String
    }
}`;

export const serverCode = `using Example.Hello; // generated by NSmithy

public class HelloHandler : IHelloServiceHandler
//                          ^? IHelloServiceHandler
{
    public Task<SayHelloOutput> SayHelloAsync(
//              ^? SayHelloOutput
        SayHelloInput input,
        CancellationToken ct = default
    )
//      ^? SayHelloInput
    {
        return Task.FromResult(new SayHelloOutput("Hello, " + input.Name + "!"));
//                                 ^? SayHelloOutput
    }
}

builder.Services.AddSmithyServer<IHelloServiceHandler, HelloHandler>();
//                               ^? IHelloServiceHandler
app.MapHelloServiceRoutes();`;

export const clientCode = `using Example.Hello; // generated by NSmithy

IHelloServiceClient client = new HelloServiceClient(transport, options);
//                  ^? IHelloServiceClient
//                               ^? HelloServiceClient

var response = await client.SayHelloAsync(new SayHelloInput("World"));
//  ^? SayHelloOutput
//                          ^? SayHelloAsync
//                                            ^? SayHelloInput
Console.WriteLine(response.Message); // Hello, World!`;

export const docsCode = `// NSmithy.AspNetCore.Docs
app.MapSmithyOpenApi(); // Scalar UI  →  /openapi
app.MapSmithyDocs();    // Sphinx HTML →  /docs`;

### Define your contract

Start by describing your API in [Smithy IDL](https://smithy.io) — operations, input and output shapes, and traits. Smithy is protocol-agnostic: the same model can drive a REST service, a gRPC service, or both at once.

</div>
<Code code={smithyCode} lang="smithy" />

<div>

### Implement server stubs

NSmithy generates a typed handler interface — one method per service operation. You fill in the business logic; routing, serialization, and protocol binding are handled by the generated ASP.NET Core adapter.

</div>
<TypeHintCode code={serverCode} lang="csharp" types={{
  'IHelloServiceHandler': 'interface IHelloServiceHandler\n{\n    Task<SayHelloOutput> SayHelloAsync(\n        SayHelloInput input,\n        CancellationToken ct = default\n    );\n}',
  'SayHelloOutput': 'record SayHelloOutput(string Message)',
  'SayHelloInput': 'record SayHelloInput(string Name)',
}} />

<div>

### Use the generated client

NSmithy generates a typed async client with one method per operation. Inputs and outputs are strongly-typed C# records — no string URLs, no manual JSON, no reflection.

</div>
<TypeHintCode code={clientCode} lang="csharp" types={{
  'IHelloServiceClient': 'interface IHelloServiceClient\n{\n    Task<SayHelloOutput> SayHelloAsync(\n        SayHelloInput input,\n        CancellationToken ct = default\n    );\n}',
  'HelloServiceClient': 'class HelloServiceClient : IHelloServiceClient',
  'SayHelloInput': 'record SayHelloInput(string Name)',
  'SayHelloOutput': 'record SayHelloOutput(string Message)',
  'SayHelloAsync': 'Task<SayHelloOutput> IHelloServiceClient.SayHelloAsync(SayHelloInput input, [CancellationToken ct = default])',
}} />

<div>

### Document your APIs

NSmithy generates an interactive [Scalar](https://scalar.com) OpenAPI explorer
and [Sphinx](https://www.sphinx-doc.org) reference documentation straight from
the Smithy model — no manual spec authoring. Both are built at compile time and
served as static files by ASP.NET Core.

</div>

</div>
